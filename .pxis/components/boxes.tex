\usepackage{tcolorbox}
\tcbuselibrary{most}

% ------------------------------------------------------------
% ARLIZ Callouts
%
% Minimal, borderless admonitions. No fill, no frame — just a
% thin colored rule on the left and a small bold label that
% sits directly above the body text. Only the accent color and
% the label differ between variants; everything else is shared.
% ------------------------------------------------------------

\tcbset{
	arlizcallout/.style={
		enhanced,
		colback=white,
		colframe=white,
		colbacktitle=white,
		boxrule=0pt,
		sharp corners,
		titlerule=0pt,
		toptitle=2pt,
		bottomtitle=2pt,
		fonttitle=\sffamily\bfseries\small,
		left=13pt,
		right=10pt,
		top=6pt,
		bottom=7pt,
		before skip=10pt,
		after skip=14pt,
		breakable,
	}
}

\newtcolorbox{notebox}[1][]{%
	arlizcallout,
	borderline west={2pt}{0pt}{arlizblue},
	coltitle=arlizblue,
	title=Note,
	#1
}

\newtcolorbox{tipbox}[1][]{%
	arlizcallout,
	borderline west={2pt}{0pt}{arlizlightblue},
	coltitle=arlizlightblue,
	title=Tip,
	#1
}

\newtcolorbox{warningbox}[1][]{%
	arlizcallout,
	borderline west={2pt}{0pt}{arliznavy},
	coltitle=arliznavy,
	title=Warning,
	#1
}

\newtcolorbox{examplebox}[1][]{%
	arlizcallout,
	borderline west={2pt}{0pt}{arlizgray},
	coltitle=arlizgray,
	title=Example,
	#1
}

\newtcolorbox{definitionbox}[1][]{%
	arlizcallout,
	borderline west={2pt}{0pt}{accent},
	coltitle=accent,
	title=Definition,
	#1
}
