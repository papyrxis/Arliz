  \kern 0.12em%

\newcommand{\headerfont}{\sffamily\small\bfseries}
\newcommand{\footerfont}{\sffamily\scriptsize}

\setlength{\headsep}{16pt}

\newcommand{\arlizHeadRule}{%
  \vbox to 0pt{%
    \vss%
    \hbox to \headwidth{%
      \textcolor{accent}{\leaders\hrule height 0.9pt\hfill}%
    }%
    \vskip 0.25pt%
    \hbox to \headwidth{%
      \textcolor{accent!35}{\leaders\hrule height 0.3pt\hfill}%
    }%
    \vskip 2pt%
  }%
}

\newcommand{\arlizFootRule}{%
  \vbox to 0pt{%
    \vskip -3pt%
    \hbox to \headwidth{%
      \textcolor{border}{\leaders\hrule height 0.5pt\hfill}%
    }%
    \vss%
  }%
}

\newcommand{\arlizLightRule}{%
  \hbox to \headwidth{%
    \textcolor{primary!50}{\leaders\hrule height 0.45pt\hfill}%
  }%
}

\fancypagestyle{main}{%
  \fancyhf{}

  \fancyhead[RO]{%
    \parbox[b]{\textwidth}{%
      \raggedleft%
      \headerfont\color{primary}%
      {\nouppercase{\rightmark}}%
    }%
  }

  \fancyhead[LO]{%
    \parbox[b]{\textwidth}{%
      \raggedright%
      \headerfont\color{accent}%
      \textsc{\PDFTitleFront}%
    }%
  }

  \fancyhead[LE]{%
    \parbox[b]{\textwidth}{%
      \raggedright%
      \headerfont\color{primary}%
      {\nouppercase{\leftmark}}%
    }%
  }

  \fancyhead[RE]{%
    \parbox[b]{\textwidth}{%
      \raggedleft%
      \headerfont\color{accent}%
      \textsc{\PDFTitleFront}%
    }%
  }

  \fancyfoot[LE]{%
    \footerfont\color{accent}%
    \textbf{\large\thepage}%
    \textcolor{border}{\,\textbar\,}%
    \textcolor{textsecondary}{\pageref{LastPage}}%
  }

  \fancyfoot[RO]{%
    \footerfont\color{accent}%
    \textbf{\large\thepage}%
    \textcolor{border}{\,\textbar\,}%
    \textcolor{textsecondary}{\pageref{LastPage}}%
  }

  \fancyfoot[RE]{%
    \footerfont\color{textsecondary}%
    \textit{Open Source}%
  }

  \fancyfoot[LO]{%
    \footerfont\color{textsecondary}%
    \textit{\the\year}%
  }

  \renewcommand{\headrulewidth}{0.6pt}
  \renewcommand{\footrulewidth}{0.4pt}
  \renewcommand{\headrule}{\hbox to\headwidth{\color{accent}\leaders\hrule height \headrulewidth\hfill}}
  \renewcommand{\footrule}{\hbox to\headwidth{\color{border}\leaders\hrule height \footrulewidth\hfill}}
}

\fancypagestyle{chapter}{%
  \fancyhf{}
  \renewcommand{\headrulewidth}{0pt}
  \renewcommand{\footrulewidth}{0.4pt}

  \fancyfoot[C]{%
    \footerfont\color{accent}%
    \decorativerule\quad%
    \textbf{\large\thepage}%
    \quad\decorativerule%
  }

  \renewcommand{\footrule}{\hbox to\headwidth{\color{accent!60}\leaders\hrule height \footrulewidth\hfill}}
}

% ============================================================
% Header: [centred: ARLIZ]
% Footer: [year]   [roman page]   [Arliz Vol. I]
% ============================================================
\fancypagestyle{frontmatter}{%
  \fancyhf{}

  \fancyhead[C]{%
    \headerfont\color{primary}%
    \textsc{arliz}%
  }

  \fancyfoot[L]{%
    \footerfont\color{textsecondary}%
    \the\year%
  }
  \fancyfoot[C]{%
    \pagenumberfont\color{textsecondary}%
    \thepage%
  }
  \fancyfoot[R]{%
    \footerfont\color{textsecondary}%
    \PDFTitleFront%
  }

  \renewcommand{\headrulewidth}{0.5pt}
  \renewcommand{\footrulewidth}{0.3pt}
  \renewcommand{\headrule}{\arlizLightRule}
  \renewcommand{\footrule}{%
    \hbox to \headwidth{%
      \textcolor{textsecondary!40}{\leaders\hrule height 0.3pt\hfill}%
    }%
  }
}

% ============================================================
% Header: [section name]             [ARLIZ]
% Footer: [year · ARLIZ]                          [9]
% ============================================================
\fancypagestyle{backmatter}{%
  \fancyhf{}

  \fancyhead[L]{%
    \parbox[b]{0.65\headwidth}{%
      \raggedright%
      \headerfont\color{primary}%
      \nouppercase{\leftmark}%
    }%
  }
  \fancyhead[R]{%
    \headerfont\color{accent}%
    \textsc{arliz}%
  }

  \fancyfoot[L]{%
    \footerfont\color{textsecondary}%
    \the\year\arlizDot\textsc{arliz}%
  }
  \fancyfoot[R]{%
    \pagenumberfont\color{primary}%
    \textbf{\thepage}%
  }

  \renewcommand{\headrulewidth}{0.8pt}
  \renewcommand{\footrulewidth}{0.4pt}
  \renewcommand{\headrule}{\arlizHeadRule}
  \renewcommand{\footrule}{\arlizFootRule}
}

% ============================================================
% PART — part title pages (no header, decorative footer)
% ============================================================

  \fancyhf{}
  \renewcommand{\headrulewidth}{0pt}
  \renewcommand{\footrulewidth}{0pt}

  \fancyfoot[C]{%
    \decorativerule\enskip%
    \pagenumberfont\color{primary}\textbf{\Large\thepage}%
    \enskip\decorativerule%
  }
}
